# --- R1 — Enumeración de directorios (dirb) ---

# --- R2 — Repositorio Git expuesto ---

# --- R3 — Métodos HTTP ---
curl -X OPTIONS -i http://phantomcorp/
nmap --script http-methods,http-enum phantomcorp

# --- R4 — Usuarios vía API ---
curl -i http://phantomcorp/api/users
./ctf status 06

# --- R5 — Fingerprinting (whatweb) y enumeración dirigida ---
whatweb http://phantomcorp
whatweb -a 3 http://phantomcorp
nmap -Pn --script http-enum phantomcorp -p80
gobuster dir -u http://phantomcorp -w /usr/share/dirb/wordlists/big.txt
# Consultas dirigidas adicionales (cabeceras, métodos, rutas default, UA, Host):

curl -sS -i -X OPTIONS http://phantomcorp/
nmap -Pn -sV -p1-10000 phantomcorp

# --- Cierre ---
./ctf submit 06 R5 'FLAG{...}'   (y envíos previos de R1 a R4)
