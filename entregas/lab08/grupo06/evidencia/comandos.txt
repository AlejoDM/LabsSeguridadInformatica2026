# --- P4 — R4: pivoting hacia la red interna ---
# Fecha de ejecución: 2026-10-28

# 1. Contexto y observación: identidad, interfaces, resolución y consulta del host interno.
# Ejecutado desde el host anfitrión para entrar como `operador` al contenedor víctima:
docker exec -u operador cyberlab-phantomcorp bash -lc "id; hostname; ip addr; echo '--- /etc/hosts ---'; cat /etc/hosts; echo '--- resolucion de phantomcorp-db ---'; getent hosts phantomcorp-db; echo '--- consulta al host interno desde la victima ---'; curl -isS --max-time 10 http://phantomcorp-db/ 2>&1"

# Resultado guardado en evidencia/salidas/r4-pivot-victima.txt:
# la víctima tiene eth0 172.21.0.3 (labnet) y eth1 172.20.0.3 (internalnet);
# phantomcorp-db resuelve a 172.20.0.2 y responde HTTP 200 con FLAG{pivot_internal_host}.

# 2. Contraste de posición: la consola atacante externa no alcanza el host interno.
# Ejecutado desde el host anfitrión, dentro de la consola atacante:
docker exec cyberlab-attacker bash -c "id; hostname; hostname -I; echo '--- resolucion de phantomcorp-db ---'; getent hosts phantomcorp-db || echo 'getent: phantomcorp-db no resuelve'; echo '--- intento de consulta al host interno desde el atacante ---'; curl -isS --max-time 10 http://phantomcorp-db/ 2>&1 || true; echo '--- contraste: alcance desde el atacante en labnet ---'; getent hosts phantomcorp; curl -sS --max-time 10 -o /dev/null -w 'TCP 80 phantomcorp: %{http_code}\n' --connect-timeout 3 http://phantomcorp/ 2>&1 || true"

# Resultado guardado en evidencia/salidas/r4-pivot-atacante.txt:
# el atacante solo tiene 172.21.0.2 (labnet), no resuelve phantomcorp-db
# (curl error 6: Could not resolve host) y sí resuelve a phantomcorp (172.21.0.3, labnet).

# 3. Presentación de la flag y estado, ejecutados desde la raíz del repositorio (Git Bash):
./ctf submit 08 R4 'FLAG{pivot_internal_host}'
./ctf status 08

# Resultado: [ OK ] Reto R4 resuelto: Pivoting al host interno.
