# atacante (cyberlab-attacker) salvo que se indique lo contrario.

# --- Linea base: corrida completa del agente de referencia (R1-R4) ---
# Ejecutado desde el host anfitrion, entrando a la consola atacante:
docker exec -e LLM_PROVIDER=mock cyberlab-attacker bash -lc "cd /repo/labs/lab09-agentes/src && python3 agente_pentest.py"
# Salida completa, sin editar, en evidencia/salidas/referencia-mock.txt

# --- P1 — R1: recon inicial ---
# La flag de R1 (FLAG{agente_primer_recon}) sale en el paso 2 del transcript
# de referencia (http_get a http://phantomcorp/), header X-Recon-Flag.

# Presentacion de la flag, ejecutada desde la raiz del repositorio:
./ctf submit 09 R1 'FLAG{...}'

# --- P2 — R2: enumeración y seguimiento de robots.txt ---
# Evidencia usada: pasos 3 y 4 de referencia-mock.txt.
# El agente solicita /robots.txt y, tras observar la directiva Disallow,
# consulta una única vez /vault-interno-x7.

# Presentación intentada con el verificador oficial desde una copia temporal
# normalizada a LF; fue rechazada. La salida completa se conserva en
# evidencia/salidas/r2-submit-rechazado.txt. No se creó progreso para R2.
./ctf submit 09 R2 'FLAG{agente_ruta_oculta}'

# --- P5 — R5: implementación del agente ---
# Fecha: 2026-10-26
# Proveedor: mock (sin API key).
# Objetivo autorizado: phantomcorp.

# 1. Ejecución del agente propio.
# Desde PowerShell, ingresar a la consola atacante:
docker exec -it cyberlab-attacker bash

# Dentro de la consola atacante:
cd /repo/entregas/lab09/grupo06/src
LLM_PROVIDER=mock PYTHONPATH=/repo/labs/lab09-agentes/src python3 agente_esqueleto.py

# 2. Guardar el transcript completo.
# Ejecutado desde PowerShell:
docker exec cyberlab-attacker bash -c "cd /repo/entregas/lab09/grupo06/src && LLM_PROVIDER=mock PYTHONPATH=/repo/labs/lab09-agentes/src python3 agente_esqueleto.py > /loot/r5-esqueleto-mock.txt 2>&1"

# Copiar la salida a la entrega:
# PowerShell:
Copy-Item .loot/r5-esqueleto-mock.txt entregas/lab09/grupo06/evidencia/salidas/r5-esqueleto-mock.txt

# 3. Resultado observado.
# Paso 6: GET /api/token -> HTTP 200.
# Paso 7: POST /api/vault -> HTTP 200.
# Flag encontrada: FLAG{agente_esqueleto_completo}.
# Paso 8: el guardrail bloqueó 8.8.8.8.
# Finalización correcta del agente.

# 4. Validación desde PowerShell.
# Se usó una copia temporal del validador para resolver
# los problemas de saltos de línea CRLF en Windows.
docker exec cyberlab-attacker bash /tmp/lab09-ctf/ctf submit 09 R5 'FLAG{agente_esqueleto_completo}'
docker exec cyberlab-attacker bash /tmp/lab09-ctf/ctf status 09

# Resultado: [ OK ] Reto R5 resuelto.
# El estado temporal 1/5 no representa el progreso total del grupo.

# --- P3 — R3: API de debug expuesta ---

# El agente se ejecutó dentro de cyberlab-attacker.
docker exec -e LLM_PROVIDER=mock cyberlab-attacker bash -lc "cd /repo/labs/lab09-agentes/src && python3 -u agente_pentest.py" | tee .loot/lab09-r3-repeticion.txt

# con el esperado por el verificador local.

# 4. Registro exitoso desde Git Bash, en la raíz del repositorio.
./ctf submit 09 R3 'FLAG{agente_api_expuesta}'

# Resultado:
# [ OK ] Reto R3 resuelto: El agente detecta la API expuesta

# 5. Comprobación del progreso.
./ctf status 09 | tee entregas/lab09/grupo06/evidencia/salidas/r3-status-confirmado.txt
